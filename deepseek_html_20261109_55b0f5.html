// ⚠️ نام کاربری و رمز عبور خودت را اینجا تغییر بده! const ADMIN_USERNAME = "ahoora"; const ADMIN_PASSWORD = "Niv@ro#1405"; const LOGIN_KEY = "nivaro_admin_logged_in"; const ORDERS_KEY = "nivaro_orders"; // 👈 این خط باید اینجا باشه (قبل از استفاده) // --- بخش لاگین --- if (localStorage.getItem(LOGIN_KEY) === "true") { showDashboard(); } document.getElementById('loginForm').addEventListener('submit', function(e) { e.preventDefault(); const user = document.getElementById('userInput').value.trim(); const pass = document.getElementById('passInput').value; if (user === ADMIN_USERNAME && pass === ADMIN_PASSWORD) { localStorage.setItem(LOGIN_KEY, "true"); showDashboard(); document.getElementById('error').style.display = 'none'; } else { document.getElementById('error').style.display = 'block'; document.getElementById('passInput').value = ''; } }); function showDashboard() { document.getElementById('loginView').style.display = 'none'; document.getElementById('dashboard').style.display = 'block'; renderOrders(); // حالا ORDERS_KEY تعریف شده و خطا نمیده } function logout() { localStorage.removeItem(LOGIN_KEY); document.getElementById('loginView').style.display = 'block'; document.getElementById('dashboard').style.display = 'none'; } // --- بخش مدیریت سفارش‌ها --- function getOrders() { const data = localStorage.getItem(ORDERS_KEY); return data ? JSON.parse(data) : []; } function saveOrders(orders) { localStorage.setItem(ORDERS_KEY, JSON.stringify(orders)); } document.getElementById('orderForm').addEventListener('submit', function(e) { e.preventDefault(); const name = document.getElementById('orderName').value; const type = document.getElementById('orderType').value; const price = document.getElementById('orderPrice').value; const orders = getOrders(); orders.push({ id: Date.now(), name: name, type: type, price: price, status: "pending" }); saveOrders(orders); renderOrders(); this.reset(); }); function renderOrders() { const orders = getOrders(); const tbody = document.getElementById('ordersBody'); if (orders.length === 0) { tbody.innerHTML = `هنوز سفارشی ثبت نشده است. اولین سفارش رو اضافه کن!`; return; } let html = ""; orders.slice().reverse().forEach((order) => { let statusHtml = ""; let statusText = ""; if(order.status === "pending") { statusHtml = "status-pending"; statusText = "⏳ در انتظار"; } if(order.status === "doing") { statusHtml = "status-doing"; statusText = "⚙️ در حال انجام"; } if(order.status === "done") { statusHtml = "status-done"; statusText = "✅ تحویل شده"; } html += ` ${order.name} ${order.type} ${order.price} تومان ${statusText} `; }); tbody.innerHTML = html; } window.changeStatus = function(id) { const orders = getOrders(); const order = orders.find(o => o.id === id); if(order) { if(order.status === "pending") order.status = "doing"; else if(order.status === "doing") order.status = "done"; else order.status = "pending"; saveOrders(orders); renderOrders(); } } window.deleteOrder = function(id) { if(confirm('آیا از حذف این سفارش مطمئن هستی؟')) { const orders = getOrders().filter(o => o.id !== id); saveOrders(orders); renderOrders(); } }
